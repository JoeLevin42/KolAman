using MongoDB.Driver;
using System.Collections.Generic;

var client = new MongoClient("mongodb://localhost:27017");
var db = client.GetDatabase("MyDatabase");
var collection = db.GetCollection<MyDocument>("MyCollection");

// new empty list every cycle
List<string> titles = new List<string>();

// take only the new
var items = await collection.Find(x => x.Status == "NEW").ToListAsync();

foreach (var item in items)
{
    // save the title object in list
    titles.Add(item.Title);

    // time by prior
    await Task.Delay(
        item.Priority == "low" ? 1000 :
        item.Priority == "hight" ? 3000 :
        2000
    );

    // status cahnged
    item.Status = "DONE";

    await collection.ReplaceOneAsync(
        x => x.Id == item.Id,
        item
    );
}

// all title of this cycle here
foreach (var title in titles)
{
    Console.WriteLine(title);
}

public class MyDocument
{
    public string Id { get; set; }
    public string Status { get; set; }
    public string Priority { get; set; }
    public string Title { get; set; }
}
